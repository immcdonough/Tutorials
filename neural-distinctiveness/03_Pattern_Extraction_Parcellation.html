4.2.1 Volume vs Surface Data

Format Description Data Points Use Case
NIfTI (volume) 3D voxels ~100-300k Traditional analysis
GIFTI (surface) Cortical mesh 32k/hemisphere Surface visualization
CIFTI Combined 91k grayordinates HCP pipeline

4.2.2 Why Surface/CIFTI?

  1. Better alignment: Cortical folding varies across people
  2. No smoothing across sulci: Volume smoothing blurs across gyri
  3. Established atlases: HCP-MMP1.0 defines 360 parcels

The Human Connectome Project Multi-Modal Parcellation defines 180 parcels per hemisphere based on:

def simulate_parcellation(betas_dense, parcel_labels):
    """Simulate parcellation by averaging within parcels."""
    n_conditions = betas_dense.shape[0]
    n_parcels = int(parcel_labels.max())

    betas_parcel = np.zeros((n_conditions, n_parcels))
    for p in range(n_parcels):
        mask = parcel_labels == (p + 1)
        if mask.sum() > 0:
            betas_parcel[:, p] = betas_dense[:, mask].mean(axis=1)

    return betas_parcel

4.4.1 Final Data Structure

For each subject:

patterns: (n_runs, n_conditions, n_parcels)
          (3, 6, 360)

residuals_per_run: list of (n_timepoints, n_parcels) arrays
def create_pattern_structure(n_runs=3, n_conditions=6, n_parcels=360,
                             n_timepoints=300, effect_size=0.5):
    """Create simulated pattern data."""
    # Face and object selective parcels
    true_face_pattern = np.zeros(n_parcels)
    true_face_pattern[50:60] = effect_size
    true_face_pattern[150:160] = effect_size

    true_object_pattern = np.zeros(n_parcels)
    true_object_pattern[70:80] = effect_size
    true_object_pattern[170:180] = effect_size

    patterns = np.zeros((n_runs, n_conditions, n_parcels))
    residuals = []

    for run in range(n_runs):
        run_noise = np.random.randn(n_parcels) * 0.1

        patterns[run, 0] = true_face_pattern + run_noise + np.random.randn(n_parcels) * 0.2
        patterns[run, 1] = true_face_pattern * 0.9 + run_noise + np.random.randn(n_parcels) * 0.2
        patterns[run, 2] = true_face_pattern * 0.7 + run_noise + np.random.randn(n_parcels) * 0.2
        patterns[run, 3] = true_object_pattern + run_noise + np.random.randn(n_parcels) * 0.2
        patterns[run, 4] = true_object_pattern * 0.9 + run_noise + np.random.randn(n_parcels) * 0.2
        patterns[run, 5] = true_object_pattern * 0.7 + run_noise + np.random.randn(n_parcels) * 0.2

        res = np.random.randn(n_timepoints, n_parcels) * 0.5
        residuals.append(res)

    return patterns, residuals

4.5 Part 4: Quality Checks

def pattern_quality_checks(patterns, residuals):
    """Quality checks on pattern data."""
    n_runs, n_conditions, n_parcels = patterns.shape
    qc = {}

    qc['n_nan'] = np.sum(np.isnan(patterns))
    qc['n_inf'] = np.sum(np.isinf(patterns))

    parcel_vars = patterns.var(axis=(0, 1))
    qc['n_zero_var'] = np.sum(parcel_vars < 1e-10)

    reliabilities = []
    for p in range(n_parcels):
        corrs = []
        for r1 in range(n_runs):
            for r2 in range(r1 + 1, n_runs):
                c = np.corrcoef(patterns[r1, :, p], patterns[r2, :, p])[0, 1]
                if not np.isnan(c):
                    corrs.append(c)
        if corrs:
            reliabilities.append(np.mean(corrs))

    qc['mean_reliability'] = np.mean(reliabilities)
    qc['n_low_reliability'] = np.sum(np.array(reliabilities) < 0.3)

    signal_var = patterns.var()
    noise_var = np.mean([r.var() for r in residuals])
    qc['snr'] = signal_var / noise_var if noise_var > 0 else np.inf

    qc['beta_range_ok'] = -10 < patterns.min() and patterns.max() < 10

    return qc

qc = pattern_quality_checks(patterns, residuals)

print("=== Quality Check Results ===")
print(f"\nData integrity:")
print(f"  NaN values: {qc['n_nan']}")
print(f"  Inf values: {qc['n_inf']}")
print(f"  Zero-variance parcels: {qc['n_zero_var']}")
print(f"\nReliability:")
print(f"  Mean cross-run: {qc['mean_reliability']:.3f}")
print(f"  Low reliability parcels: {qc['n_low_reliability']}")
print(f"\nSignal quality:")
print(f"  SNR: {qc['snr']:.3f}")
print(f"  Beta range OK: {qc['beta_range_ok']}")

def reorder_conditions(patterns, condition_names, new_order):
    """Reorder conditions in the pattern array."""
    if isinstance(new_order[0], str):
        new_order = [list(condition_names).index(n) for n in new_order]
    return patterns[:, new_order, :], condition_names[new_order]

## Part 1: Understanding CIFTI Files

### Volume vs Surface Data

| Format | Description | Data Points | Use Case |
|--------|-------------|-------------|----------|
| NIfTI (volume) | 3D voxels | ~100-300k | Traditional analysis |
| GIFTI (surface) | Cortical mesh | 32k/hemisphere | Surface visualization |
| **CIFTI** | Combined | **91k grayordinates** | **HCP pipeline** |

### Why Surface/CIFTI?

1. **Better alignment**: Cortical folding varies across people
2. **No smoothing across sulci**: Volume smoothing blurs across gyri
3. **Established atlases**: HCP-MMP1.0 defines 360 parcels

### CIFTI File Types

| Extension | Description | Shape |
|-----------|-------------|-------|
| `.dtseries.nii` | Dense timeseries | (timepoints, grayordinates) |
| `.dscalar.nii` | Dense scalars | (maps, grayordinates) |
| `.dlabel.nii` | Dense labels | (1, grayordinates) |
| `.ptseries.nii` | Parcellated timeseries | (timepoints, parcels) |
| `.pscalar.nii` | Parcellated scalars | (maps, parcels) |

```{python}
def create_pattern_structure(n_runs=3, n_conditions=6, n_parcels=360,
                             n_timepoints=300, effect_size=0.5):
    """Create simulated pattern data."""
    # Face and object selective parcels
    true_face_pattern = np.zeros(n_parcels)
    true_face_pattern[50:60] = effect_size
    true_face_pattern[150:160] = effect_size

    true_object_pattern = np.zeros(n_parcels)
    true_object_pattern[70:80] = effect_size
    true_object_pattern[170:180] = effect_size

    patterns = np.zeros((n_runs, n_conditions, n_parcels))
    residuals = []

    for run in range(n_runs):
        run_noise = np.random.randn(n_parcels) * 0.1

        patterns[run, 0] = true_face_pattern + run_noise + np.random.randn(n_parcels) * 0.2
        patterns[run, 1] = true_face_pattern * 0.9 + run_noise + np.random.randn(n_parcels) * 0.2
        patterns[run, 2] = true_face_pattern * 0.7 + run_noise + np.random.randn(n_parcels) * 0.2
        patterns[run, 3] = true_object_pattern + run_noise + np.random.randn(n_parcels) * 0.2
        patterns[run, 4] = true_object_pattern * 0.9 + run_noise + np.random.randn(n_parcels) * 0.2
        patterns[run, 5] = true_object_pattern * 0.7 + run_noise + np.random.randn(n_parcels) * 0.2

        res = np.random.randn(n_timepoints, n_parcels) * 0.5
        residuals.append(res)

    return patterns, residuals
```

## Part 4: Quality Checks

```{python}
#| label: qc-checks

def pattern_quality_checks(patterns, residuals):
    """Quality checks on pattern data."""
    n_runs, n_conditions, n_parcels = patterns.shape
    qc = {}

    qc['n_nan'] = np.sum(np.isnan(patterns))
    qc['n_inf'] = np.sum(np.isinf(patterns))

    parcel_vars = patterns.var(axis=(0, 1))
    qc['n_zero_var'] = np.sum(parcel_vars < 1e-10)

    reliabilities = []
    for p in range(n_parcels):
        corrs = []
        for r1 in range(n_runs):
            for r2 in range(r1 + 1, n_runs):
                c = np.corrcoef(patterns[r1, :, p], patterns[r2, :, p])[0, 1]
                if not np.isnan(c):
                    corrs.append(c)
        if corrs:
            reliabilities.append(np.mean(corrs))

    qc['mean_reliability'] = np.mean(reliabilities)
    qc['n_low_reliability'] = np.sum(np.array(reliabilities) < 0.3)

    signal_var = patterns.var()
    noise_var = np.mean([r.var() for r in residuals])
    qc['snr'] = signal_var / noise_var if noise_var > 0 else np.inf

    qc['beta_range_ok'] = -10 < patterns.min() and patterns.max() < 10

    return qc

qc = pattern_quality_checks(patterns, residuals)

print("=== Quality Check Results ===")
print(f"\nData integrity:")
print(f"  NaN values: {qc['n_nan']}")
print(f"  Inf values: {qc['n_inf']}")
print(f"  Zero-variance parcels: {qc['n_zero_var']}")
print(f"\nReliability:")
print(f"  Mean cross-run: {qc['mean_reliability']:.3f}")
print(f"  Low reliability parcels: {qc['n_low_reliability']}")
print(f"\nSignal quality:")
print(f"  SNR: {qc['snr']:.3f}")
print(f"  Beta range OK: {qc['beta_range_ok']}")
```

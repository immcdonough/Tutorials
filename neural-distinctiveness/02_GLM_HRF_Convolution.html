3.2 Part 1: The Hemodynamic Response Function (HRF)

3.2.1 Why Can’t We Just Use Raw Stimulus Timings?

fMRI measures BOLD signal (Blood Oxygen Level Dependent), which reflects:

  • Blood flow changes
  • Blood oxygenation changes

This response is slow and delayed compared to neural activity:

Process Timing
Neural activity ~100 ms
BOLD peak ~5-6 seconds
Return to baseline ~15-20 seconds

The HRF models this sluggish hemodynamic response.

def spm_hrf(tr, oversampling=16, time_length=32, peak_delay=6,
            undershoot_delay=16, peak_dispersion=1, undershoot_dispersion=1,
            peak_undershoot_ratio=6):
    """Generate SPM canonical HRF."""
    dt = tr / oversampling
    time_stamps = np.arange(0, time_length, dt)

    # Gamma function parameters
    peak = time_stamps ** (peak_delay / peak_dispersion - 1) * \
           np.exp(-time_stamps / peak_dispersion) / \
           (gamma_func(peak_delay / peak_dispersion) *
            peak_dispersion ** (peak_delay / peak_dispersion))

    undershoot = time_stamps ** (undershoot_delay / undershoot_dispersion - 1) * \
                 np.exp(-time_stamps / undershoot_dispersion) / \
                 (gamma_func(undershoot_delay / undershoot_dispersion) *
                  undershoot_dispersion ** (undershoot_delay / undershoot_dispersion))

    hrf_oversampled = peak - undershoot / peak_undershoot_ratio
    hrf_oversampled = hrf_oversampled / np.max(hrf_oversampled)
    hrf = hrf_oversampled[::oversampling]

    expected_len = int(time_length / tr)
    if len(hrf) > expected_len:
        hrf = hrf[:expected_len]

    return hrf, time_stamps[::oversampling][:len(hrf)]

3.3 Part 2: Convolution - Predicting the BOLD Signal

Given stimulus onsets, we convolve with the HRF to predict the expected BOLD signal:

\[\text{predicted\_BOLD}(t) = \sum_{\tau} \text{stimulus}(\tau) \times \text{HRF}(t - \tau)\]

def create_stimulus_timecourse(onsets, n_timepoints, tr):
    """Create binary stimulus timecourse from onset times."""
    stim = np.zeros(n_timepoints)
    for onset in onsets:
        tr_idx = int(np.round(onset / tr))
        if 0 <= tr_idx < n_timepoints:
            stim[tr_idx] = 1
    return stim

def convolve_with_hrf(stimulus_timecourse, hrf):
    """Convolve stimulus timecourse with HRF."""
    convolved = np.convolve(stimulus_timecourse, hrf, mode='full')
    return convolved[:len(stimulus_timecourse)]

3.4 Part 3: Building the Complete Design Matrix

The GLM design matrix includes:

  1. Task regressors (HRF-convolved) - one per condition
  2. Motion parameters - translation and rotation (6 params)
  3. High-pass filter basis - removes slow drifts
  4. Intercept - overall mean
def create_dct_highpass_filter(n_timepoints, tr, cutoff=128):
    """Create DCT basis for high-pass filtering."""
    total_time = n_timepoints * tr
    n_basis = int(np.floor(2 * total_time / cutoff) + 1)

    dct_basis = np.zeros((n_timepoints, n_basis))
    time_idx = np.arange(n_timepoints)

    for k in range(n_basis):
        dct_basis[:, k] = np.cos(np.pi * k * (2 * time_idx + 1) / (2 * n_timepoints))

    return dct_basis

def build_design_matrix(timing_dict, n_timepoints, tr, hrf,
                        motion_params=None, highpass_cutoff=128):
    """Build complete GLM design matrix."""
    columns = []
    names = []

    # Task regressors
    for cond_name, onsets in timing_dict.items():
        stim = create_stimulus_timecourse(onsets, n_timepoints, tr)
        reg = convolve_with_hrf(stim, hrf)
        columns.append(reg)
        names.append(cond_name)

    # Motion parameters
    if motion_params is not None:
        motion_names = ['trans_x', 'trans_y', 'trans_z', 'rot_x', 'rot_y', 'rot_z']
        for i, name in enumerate(motion_names):
            columns.append(motion_params[:, i])
            names.append(name)

    # High-pass filter
    dct_basis = create_dct_highpass_filter(n_timepoints, tr, highpass_cutoff)
    for i in range(dct_basis.shape[1]):
        columns.append(dct_basis[:, i])
        names.append(f'dct_{i}')

    # Intercept
    columns.append(np.ones(n_timepoints))
    names.append('intercept')

    X = np.column_stack(columns)
    return X, names

3.5 Part 4: Fitting the GLM

3.5.1 The GLM Equation

\[\mathbf{Y} = \mathbf{X}\boldsymbol{\beta} + \boldsymbol{\epsilon}\]

3.5.2 OLS Solution

\[\hat{\boldsymbol{\beta}} = (\mathbf{X}^T\mathbf{X})^{-1}\mathbf{X}^T\mathbf{Y}\]

def fit_glm_ols(X, Y):
    """Fit GLM using Ordinary Least Squares."""
    XtX_inv = np.linalg.pinv(X.T @ X)
    betas = XtX_inv @ X.T @ Y
    Y_pred = X @ betas
    residuals = Y - Y_pred
    return betas, residuals

3.6 Part 5: Robust Regression (Advanced)

TipWhen to Use Robust Regression
Method Speed Use Case
OLS Fast (~2 sec/run) Good data, quick analysis
Robust IRLS Slow (~1 hour/run) Noisy data, motion artifacts 
def huber_weights(residuals, k=1.345):
    """Compute Huber weights for robust regression."""
    mad = np.median(np.abs(residuals - np.median(residuals)))
    scale = mad / 0.6745
    u = residuals / (scale + 1e-10)
    weights = np.where(np.abs(u) <= k, 1, k / np.abs(u))
    return weights

def fit_glm_robust(X, Y, max_iter=50, tol=1e-6):
    """Fit GLM using IRLS."""
    n_timepoints, n_voxels = Y.shape
    n_regressors = X.shape[1]

    betas, residuals = fit_glm_ols(X, Y)
    weights = np.ones((n_timepoints, n_voxels))

    for iteration in range(max_iter):
        betas_old = betas.copy()

        for v in range(n_voxels):
            weights[:, v] = huber_weights(residuals[:, v])

        for v in range(n_voxels):
            W = np.diag(weights[:, v])
            XtWX = X.T @ W @ X
            XtWY = X.T @ W @ Y[:, v]
            betas[:, v] = np.linalg.solve(XtWX + 1e-8 * np.eye(n_regressors), XtWY)

        residuals = Y - X @ betas
        if np.max(np.abs(betas - betas_old)) < tol:
            break

    return betas, weights

## Part 1: The Hemodynamic Response Function (HRF)

### Why Can't We Just Use Raw Stimulus Timings?

fMRI measures **BOLD signal** (Blood Oxygen Level Dependent), which reflects:

- Blood flow changes
- Blood oxygenation changes

This response is **slow and delayed** compared to neural activity:

| Process | Timing |
|---------|--------|
| Neural activity | ~100 ms |
| BOLD peak | ~5-6 seconds |
| Return to baseline | ~15-20 seconds |

The **HRF** models this sluggish hemodynamic response.

```{python}
#| label: hrf-demo

def spm_hrf(tr, oversampling=16, time_length=32, peak_delay=6,
            undershoot_delay=16, peak_dispersion=1, undershoot_dispersion=1,
            peak_undershoot_ratio=6):
    """Generate SPM canonical HRF."""
    dt = tr / oversampling
    time_stamps = np.arange(0, time_length, dt)

    # Gamma function parameters
    peak = time_stamps ** (peak_delay / peak_dispersion - 1) * \
           np.exp(-time_stamps / peak_dispersion) / \
           (gamma_func(peak_delay / peak_dispersion) *
            peak_dispersion ** (peak_delay / peak_dispersion))

    undershoot = time_stamps ** (undershoot_delay / undershoot_dispersion - 1) * \
                 np.exp(-time_stamps / undershoot_dispersion) / \
                 (gamma_func(undershoot_delay / undershoot_dispersion) *
                  undershoot_dispersion ** (undershoot_delay / undershoot_dispersion))

    hrf_oversampled = peak - undershoot / peak_undershoot_ratio
    hrf_oversampled = hrf_oversampled / np.max(hrf_oversampled)
    hrf = hrf_oversampled[::oversampling]

    expected_len = int(time_length / tr)
    if len(hrf) > expected_len:
        hrf = hrf[:expected_len]

    return hrf, time_stamps[::oversampling][:len(hrf)]
```

## Part 2: Convolution - Predicting the BOLD Signal

Given stimulus onsets, we **convolve** with the HRF to predict the expected BOLD signal:

$$\text{predicted\_BOLD}(t) = \sum_{\tau} \text{stimulus}(\tau) \times \text{HRF}(t - \tau)$$

```{python}
#| label: convolution-demo
#| fig-cap: "From stimulus onsets to predicted BOLD response"

def create_stimulus_timecourse(onsets, n_timepoints, tr):
    """Create binary stimulus timecourse from onset times."""
    stim = np.zeros(n_timepoints)
    for onset in onsets:
        tr_idx = int(np.round(onset / tr))
        if 0 <= tr_idx < n_timepoints:
            stim[tr_idx] = 1
    return stim

def convolve_with_hrf(stimulus_timecourse, hrf):
    """Convolve stimulus timecourse with HRF."""
    convolved = np.convolve(stimulus_timecourse, hrf, mode='full')
    return convolved[:len(stimulus_timecourse)]

# Simulate a run
```

## Part 3: Building the Complete Design Matrix

The GLM design matrix includes:

1. **Task regressors** (HRF-convolved) - one per condition
2. **Motion parameters** - translation and rotation (6 params)
3. **High-pass filter basis** - removes slow drifts
4. **Intercept** - overall mean

```{python}
#| label: design-matrix
#| fig-cap: "Complete GLM design matrix structure"

def create_dct_highpass_filter(n_timepoints, tr, cutoff=128):
    """Create DCT basis for high-pass filtering."""
    total_time = n_timepoints * tr
    n_basis = int(np.floor(2 * total_time / cutoff) + 1)

    dct_basis = np.zeros((n_timepoints, n_basis))
    time_idx = np.arange(n_timepoints)

    for k in range(n_basis):
        dct_basis[:, k] = np.cos(np.pi * k * (2 * time_idx + 1) / (2 * n_timepoints))

    return dct_basis

def build_design_matrix(timing_dict, n_timepoints, tr, hrf,
                        motion_params=None, highpass_cutoff=128):
    """Build complete GLM design matrix."""
    columns = []
    names = []

    # Task regressors
    for cond_name, onsets in timing_dict.items():
        stim = create_stimulus_timecourse(onsets, n_timepoints, tr)
        reg = convolve_with_hrf(stim, hrf)
        columns.append(reg)
        names.append(cond_name)

    # Motion parameters
    if motion_params is not None:
        motion_names = ['trans_x', 'trans_y', 'trans_z', 'rot_x', 'rot_y', 'rot_z']
        for i, name in enumerate(motion_names):
            columns.append(motion_params[:, i])
            names.append(name)

    # High-pass filter
    dct_basis = create_dct_highpass_filter(n_timepoints, tr, highpass_cutoff)
    for i in range(dct_basis.shape[1]):
        columns.append(dct_basis[:, i])
        names.append(f'dct_{i}')

    # Intercept
    columns.append(np.ones(n_timepoints))
    names.append('intercept')

    X = np.column_stack(columns)
    return X, names

# Create timing for 6 conditions
conditions = ['FaceSingle', 'FaceRepeatFirst', 'FaceRepeatSubsequent',
              'ObjectSingle', 'ObjectRepeatFirst', 'ObjectRepeatSubsequent']

print(f"  - Intercept: 1")
```

## Part 4: Fitting the GLM

### The GLM Equation

$$\mathbf{Y} = \mathbf{X}\boldsymbol{\beta} + \boldsymbol{\epsilon}$$

### OLS Solution

$$\hat{\boldsymbol{\beta}} = (\mathbf{X}^T\mathbf{X})^{-1}\mathbf{X}^T\mathbf{Y}$$

```{python}
#| label: glm-fit
#| fig-cap: "GLM beta estimation and recovery"

def fit_glm_ols(X, Y):
    """Fit GLM using Ordinary Least Squares."""
    XtX_inv = np.linalg.pinv(X.T @ X)
    betas = XtX_inv @ X.T @ Y
    Y_pred = X @ betas
    residuals = Y - Y_pred
    return betas, residuals

# Simulate fMRI data

true_betas[4, 50:] = 1.3
true_betas[5, 50:] = 1.0
```

## Part 5: Robust Regression (Advanced)

::: {.callout-tip}
## When to Use Robust Regression

| Method | Speed | Use Case |
|--------|-------|----------|
| OLS | Fast (~2 sec/run) | Good data, quick analysis |
| Robust IRLS | Slow (~1 hour/run) | Noisy data, motion artifacts |
:::

```{python}
#| label: robust-demo
#| fig-cap: "Comparing OLS and robust regression with outliers"

def huber_weights(residuals, k=1.345):
    """Compute Huber weights for robust regression."""
    mad = np.median(np.abs(residuals - np.median(residuals)))
    scale = mad / 0.6745
    u = residuals / (scale + 1e-10)
    weights = np.where(np.abs(u) <= k, 1, k / np.abs(u))
    return weights

def fit_glm_robust(X, Y, max_iter=50, tol=1e-6):
    """Fit GLM using IRLS."""
    n_timepoints, n_voxels = Y.shape
    n_regressors = X.shape[1]

    betas, residuals = fit_glm_ols(X, Y)
    weights = np.ones((n_timepoints, n_voxels))

    for iteration in range(max_iter):
        betas_old = betas.copy()

        for v in range(n_voxels):
            weights[:, v] = huber_weights(residuals[:, v])

        for v in range(n_voxels):
            W = np.diag(weights[:, v])
            XtWX = X.T @ W @ X
            XtWY = X.T @ W @ Y[:, v]
            betas[:, v] = np.linalg.solve(XtWX + 1e-8 * np.eye(n_regressors), XtWY)

        residuals = Y - X @ betas
        if np.max(np.abs(betas - betas_old)) < tol:
            break

    return betas, weights

# Add outliers
Y_outliers = Y.copy()
n_outliers = 10
outlier_times = np.random.choice(n_timepoints, n_outliers, replace=False)
Y_outliers[outlier_times, :] += np.random.randn(n_outliers, n_voxels) * 20

# Fit both methods
betas_ols, _ = fit_glm_ols(X, Y_outliers)
betas_robust, weights = fit_glm_robust(X, Y_outliers)

# Compare

axes[0, 0].scatter(time[outlier_times], Y_outliers[outlier_times, 0],
                   c='red', s=100, zorder=5, label='Outliers')
axes[0, 0].set_xlabel('Time (seconds)')
axes[0, 0].set_ylabel('Signal')
axes[0, 0].set_title('Example Voxel with Spike Artifacts')
axes[0, 0].legend()

axes[0, 1].plot(time, weights[:, 0], 'g-', linewidth=1.5)
axes[0, 1].scatter(time[outlier_times], weights[outlier_times, 0], c='red', s=100)
axes[0, 1].set_xlabel('Time (seconds)')
axes[0, 1].set_ylabel('Weight')
axes[0, 1].set_title('Robust Weights (outliers downweighted)')
axes[0, 1].set_ylim(-0.1, 1.1)

corr_ols = [np.corrcoef(true_betas[i], betas_ols[i])[0, 1] for i in range(6)]
corr_robust = [np.corrcoef(true_betas[i], betas_robust[i])[0, 1] for i in range(6)]

axes[1, 0].set_xticks(range(6))
axes[1, 0].set_xticklabels(['FS', 'FRF', 'FRS', 'OS', 'ORF', 'ORS'])
axes[1, 0].set_ylabel('Correlation')
axes[1, 0].set_title(f'OLS Recovery (mean r = {np.mean(corr_ols):.3f})')
axes[1, 0].set_ylim(0, 1)

axes[1, 1].bar(range(6), corr_robust, color='steelblue', alpha=0.8)
axes[1, 1].set_xticks(range(6))
```

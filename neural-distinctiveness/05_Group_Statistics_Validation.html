def simulate_group_data(n_subjects=30, n_parcels=100,
                        effect_regions=None, effect_size=0.4):
    """Simulate group-level distinctiveness data."""
    if effect_regions is None:
        effect_regions = list(range(20, 40)) + list(range(60, 80))

    data = np.zeros((n_subjects, n_parcels))
    true_effects = np.zeros(n_parcels)

    for parcel in range(n_parcels):
        if parcel in effect_regions:
            true_effects[parcel] = 1
            data[:, parcel] = np.random.randn(n_subjects) + effect_size
        else:
            data[:, parcel] = np.random.randn(n_subjects)

    return data, true_effects

def one_sample_ttest(data, test_value=0):
    """Perform one-sample t-test for each parcel."""
    n_subjects, n_parcels = data.shape
    t_stats = np.zeros(n_parcels)
    p_values = np.zeros(n_parcels)

    for parcel in range(n_parcels):
        t, p = stats.ttest_1samp(data[:, parcel], test_value)
        t_stats[parcel] = t
        p_values[parcel] = p

    return t_stats, p_values

6.3 Part 2: Multiple Comparison Correction (FDR)

6.3.1 The Problem

With 360 parcels at α = 0.05:

  • Expected false positives = 360 × 0.05 = 18 parcels!

6.3.2 FDR: Benjamini-Hochberg Procedure

  1. Sort p-values: p₁ ≤ p₂ ≤ … ≤ pₘ
  2. Find largest k where pₖ ≤ (k/m) × q
  3. Reject all hypotheses with p ≤ pₖ
def fdr_correction(p_values, q=0.05):
    """Apply FDR correction (Benjamini-Hochberg)."""
    if HAS_STATSMODELS:
        reject, q_values, _, _ = multipletests(p_values, alpha=q, method='fdr_bh')
    else:
        m = len(p_values)
        sorted_idx = np.argsort(p_values)
        sorted_p = p_values[sorted_idx]

        thresholds = np.arange(1, m + 1) / m * q
        below = sorted_p <= thresholds

        if below.any():
            max_k = np.max(np.where(below)[0])
            reject_sorted = np.zeros(m, dtype=bool)
            reject_sorted[:max_k + 1] = True
        else:
            reject_sorted = np.zeros(m, dtype=bool)

        reject = np.zeros(m, dtype=bool)
        reject[sorted_idx] = reject_sorted

        q_values = np.minimum(sorted_p * m / np.arange(1, m + 1), 1)
        q_values = np.minimum.accumulate(q_values[::-1])[::-1]
        q_values_unsorted = np.zeros(m)
        q_values_unsorted[sorted_idx] = q_values
        q_values = q_values_unsorted

    return reject, q_values

6.4 Part 3: Effect Sizes

P-values tell us whether there’s an effect, not how big.

def compute_effect_sizes(data):
    """Compute Cohen's d and confidence intervals."""
    n_subjects, n_parcels = data.shape
    means = data.mean(axis=0)
    stds = data.std(axis=0, ddof=1)
    cohens_d = means / stds

    sem = stds / np.sqrt(n_subjects)
    t_crit = stats.t.ppf(0.975, n_subjects - 1)
    ci_lower = means - t_crit * sem
    ci_upper = means + t_crit * sem

    return cohens_d, ci_lower, ci_upper

6.5 Part 4: Reliability Validation

def split_half_reliability(data, n_iterations=100):
    """Compute split-half reliability across subjects."""
    n_subjects, n_parcels = data.shape
    half = n_subjects // 2
    correlations = []

    for _ in range(n_iterations):
        perm = np.random.permutation(n_subjects)
        group1 = perm[:half]
        group2 = perm[half:2*half]

        mean1 = data[group1].mean(axis=0)
        mean2 = data[group2].mean(axis=0)
        r = np.corrcoef(mean1, mean2)[0, 1]
        correlations.append(r)

    raw_r = np.mean(correlations)
    # Spearman-Brown correction
    reliability = 2 * raw_r / (1 + raw_r)

    return reliability, np.array(correlations)

def run_sanity_checks(data, t_stats, p_values, reject_fdr):
    """Run comprehensive sanity checks."""
    n_subjects, n_parcels = data.shape
    checks = {}

    checks['has_significant'] = reject_fdr.sum() > 0
    checks['n_significant'] = reject_fdr.sum()

    parcel_means = data.mean(axis=0)
    checks['prop_positive'] = (parcel_means > 0).mean()
    checks['positive_check'] = checks['prop_positive'] > 0.5

    checks['mean_d_single'] = parcel_means.mean()
    checks['range_reasonable'] = -2 < parcel_means.min() and parcel_means.max() < 5

    checks['no_nan'] = not np.any(np.isnan(data))
    checks['no_inf'] = not np.any(np.isinf(data))

    checks['t_not_extreme'] = -10 < t_stats.min() and t_stats.max() < 20

    return checks

checks = run_sanity_checks(data, t_stats, p_values, reject_fdr)

print("=== Sanity Checks ===")

    if isinstance(result, bool):
        status = "PASS" if result else "FAIL"
    else:
        status = f"{result:.3f}" if isinstance(result, float) else str(result)
    print(f"  {name}: {status}")

def simulate_power(effect_size, n_subjects, n_simulations=500):
    """Estimate power via simulation."""
    significant = 0
    for _ in range(n_simulations):
        data_sim = np.random.randn(n_subjects) + effect_size
        _, p = stats.ttest_1samp(data_sim, 0)
        if p < 0.05:
            significant += 1
    return significant / n_simulations

## Part 1: Group-Level Hypothesis Testing

### The Question

For each parcel, test:

- **H₀**: D_single = 0 (no category distinction)
- **H₁**: D_single > 0 (distinct representations)

```{python}
#| label: simulate-group

def simulate_group_data(n_subjects=30, n_parcels=100,
                        effect_regions=None, effect_size=0.4):
    """Simulate group-level distinctiveness data."""
    if effect_regions is None:
        effect_regions = list(range(20, 40)) + list(range(60, 80))

    data = np.zeros((n_subjects, n_parcels))
    true_effects = np.zeros(n_parcels)

    for parcel in range(n_parcels):
        if parcel in effect_regions:
            true_effects[parcel] = 1
            data[:, parcel] = np.random.randn(n_subjects) + effect_size
        else:
            data[:, parcel] = np.random.randn(n_subjects)

    return data, true_effects
```

## Part 2: Multiple Comparison Correction (FDR)

### The Problem

With 360 parcels at α = 0.05:

- Expected false positives = 360 × 0.05 = **18 parcels!**

### FDR: Benjamini-Hochberg Procedure

1. Sort p-values: p₁ ≤ p₂ ≤ ... ≤ pₘ
2. Find largest k where pₖ ≤ (k/m) × q
3. Reject all hypotheses with p ≤ pₖ

```{python}
#| label: fdr
#| fig-cap: "FDR correction comparison"

def fdr_correction(p_values, q=0.05):
    """Apply FDR correction (Benjamini-Hochberg)."""
    if HAS_STATSMODELS:
        reject, q_values, _, _ = multipletests(p_values, alpha=q, method='fdr_bh')
    else:
        m = len(p_values)
        sorted_idx = np.argsort(p_values)
        sorted_p = p_values[sorted_idx]

        thresholds = np.arange(1, m + 1) / m * q
        below = sorted_p <= thresholds

        if below.any():
            max_k = np.max(np.where(below)[0])
            reject_sorted = np.zeros(m, dtype=bool)
            reject_sorted[:max_k + 1] = True
        else:
            reject_sorted = np.zeros(m, dtype=bool)

        reject = np.zeros(m, dtype=bool)
        reject[sorted_idx] = reject_sorted

        q_values = np.minimum(sorted_p * m / np.arange(1, m + 1), 1)
        q_values = np.minimum.accumulate(q_values[::-1])[::-1]
        q_values_unsorted = np.zeros(m)
        q_values_unsorted[sorted_idx] = q_values
        q_values = q_values_unsorted

    return reject, q_values
```

```{python}
#| fig-cap: "Cohen's d effect sizes per parcel"

def compute_effect_sizes(data):
    """Compute Cohen's d and confidence intervals."""
    n_subjects, n_parcels = data.shape
    means = data.mean(axis=0)
    stds = data.std(axis=0, ddof=1)
    cohens_d = means / stds

    sem = stds / np.sqrt(n_subjects)
    t_crit = stats.t.ppf(0.975, n_subjects - 1)
    ci_lower = means - t_crit * sem
    ci_upper = means + t_crit * sem

    return cohens_d, ci_lower, ci_upper
```

```{python}
def split_half_reliability(data, n_iterations=100):
    """Compute split-half reliability across subjects."""
    n_subjects, n_parcels = data.shape
    half = n_subjects // 2
    correlations = []

    for _ in range(n_iterations):
        perm = np.random.permutation(n_subjects)
        group1 = perm[:half]
        group2 = perm[half:2*half]

        mean1 = data[group1].mean(axis=0)
        mean2 = data[group2].mean(axis=0)
        r = np.corrcoef(mean1, mean2)[0, 1]
        correlations.append(r)

    raw_r = np.mean(correlations)
    # Spearman-Brown correction
    reliability = 2 * raw_r / (1 + raw_r)

    return reliability, np.array(correlations)

reliability, split_corrs = split_half_reliability(data)
```

```{python}
#| label: ex-power

def simulate_power(effect_size, n_subjects, n_simulations=500):
    """Estimate power via simulation."""
    significant = 0
    for _ in range(n_simulations):
        data_sim = np.random.randn(n_subjects) + effect_size
        _, p = stats.ttest_1samp(data_sim, 0)
        if p < 0.05:
            significant += 1
    return significant / n_simulations

sample_sizes = [10, 15, 20, 25, 30, 35, 40, 50]
powers = [simulate_power(0.5, n) for n in sample_sizes]

plt.grid(True)
```
